# pico-atom — one source tree, two targets (design.md §14):
#   - the UF2 for the PicoCalc (default)
#   - a host test binary that compiles src/core/ with the system compiler
#     and no Pico SDK, behind -DPICO_ATOM_HOST=ON, run under CTest.
#
# The core building clean under -Wall -Wextra -Werror on both toolchains
# is the mechanism that keeps SDK dependencies from leaking downwards.

cmake_minimum_required(VERSION 3.13)

option(PICO_ATOM_HOST "Build the portable core and its host tests, with no Pico SDK" OFF)

# Every fixed capacity lives in src/core/config.h; nothing here duplicates one.
set(PICO_ATOM_CORE_SOURCES
    src/core/atom.c
    src/core/beeper.c
    src/core/bus.c
    src/core/cassette.c
    src/core/i8255.c
    src/core/i8271.c
    src/core/inflate.c
    src/core/keylayout.c
    src/core/keymap_picocalc.c
    src/core/keymatrix.c
    src/core/m6502.c
    src/core/mc6847.c
    src/core/romset.c
    src/core/sha1.c
    src/core/snappool.c
    src/core/snapshot.c
    src/core/tape.c
    src/core/uef.c
    src/core/via6522.c
)

# The MC6847 character ROM is not in the tree: §16 requires it be
# transcribed from a primary source and verified by golden image, so it
# is generated by tools/mkfont.py and picked up here once it exists.
# Without it, alpha mode draws placeholder cells and says so at boot.
if (EXISTS ${CMAKE_CURRENT_LIST_DIR}/src/core/mc6847_font.c)
    set(PICO_ATOM_HAVE_FONT 1)
    list(APPEND PICO_ATOM_CORE_SOURCES src/core/mc6847_font.c)
    message(STATUS "pico-atom: MC6847 character ROM found")
else()
    set(PICO_ATOM_HAVE_FONT 0)
    message(STATUS "pico-atom: no MC6847 character ROM "
                   "(run tools/mkfont.py; alpha mode will show placeholders)")
endif()

set(PICO_ATOM_WARNINGS -Wall -Wextra -Werror)

if (PICO_ATOM_HOST)
    project(pico_atom_host C)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    set(CMAKE_C_EXTENSIONS OFF)

    add_library(pico_atom_core STATIC ${PICO_ATOM_CORE_SOURCES})
    target_include_directories(pico_atom_core PUBLIC src/core)
    target_compile_options(pico_atom_core PRIVATE ${PICO_ATOM_WARNINGS})
    target_compile_definitions(pico_atom_core PUBLIC
        PICO_ATOM_HAVE_FONT=${PICO_ATOM_HAVE_FONT})

    enable_testing()
    add_subdirectory(test/host)
else()
    # Primary target: Raspberry Pi Pico 2 (RP2350A) — design.md §3.1.
    if (NOT DEFINED PICO_BOARD)
        set(PICO_BOARD pico2)
    endif()
    if (NOT DEFINED PICO_PLATFORM AND PICO_BOARD STREQUAL "pico2")
        set(PICO_PLATFORM rp2350)
    endif()

    include(cmake/pico_sdk_import.cmake)

    project(pico_atom C CXX ASM)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_CXX_STANDARD 17)

    pico_sdk_init()

    # FatFs is not in the tree. The SDK carries a copy for TinyUSB, and
    # ff.h includes "ffconf.h" from its own directory, so the three files
    # are copied into the build tree beside our src/port/fatfs/ffconf.h
    # rather than compiled in place (which would find TinyUSB's config).
    set(PICO_ATOM_FATFS_SRC ${PICO_SDK_PATH}/lib/tinyusb/lib/fatfs/source)
    set(PICO_ATOM_FATFS_DIR ${CMAKE_BINARY_DIR}/fatfs)
    foreach(f ff.c ff.h ffunicode.c diskio.h)
        if (NOT EXISTS ${PICO_ATOM_FATFS_SRC}/${f})
            message(FATAL_ERROR "pico-atom: FatFs ${f} not found in ${PICO_ATOM_FATFS_SRC}")
        endif()
        configure_file(${PICO_ATOM_FATFS_SRC}/${f} ${PICO_ATOM_FATFS_DIR}/${f} COPYONLY)
    endforeach()
    configure_file(src/port/fatfs/ffconf.h ${PICO_ATOM_FATFS_DIR}/ffconf.h COPYONLY)

    add_library(pico_atom_fatfs STATIC
        ${PICO_ATOM_FATFS_DIR}/ff.c
        ${PICO_ATOM_FATFS_DIR}/ffunicode.c
    )
    target_include_directories(pico_atom_fatfs PUBLIC ${PICO_ATOM_FATFS_DIR})
    # Third-party code: built as it comes, without our -Werror.

    add_executable(pico-atom
        src/port/main.c
        src/port/menu.c
        src/port/snapio.c
        src/port/textpage.c
        src/port/audio.c
        src/port/board.c
        src/port/diskio.c
        src/port/discio.c
        src/port/display.c
        src/port/kbd.c
        src/port/keymapio.c
        src/port/lcd.c
        src/port/log.c
        src/port/roms.c
        src/port/sd.c
        src/port/southbridge.c
        src/port/storage.c
        src/port/tapeio.c
        ${PICO_ATOM_CORE_SOURCES}
    )
    target_include_directories(pico-atom PRIVATE src/core src/port)
    target_compile_options(pico-atom PRIVATE ${PICO_ATOM_WARNINGS})

    # PICO_ATOM_AUDIO=OFF paces the guest on the microsecond timer instead
    # of the PCM queue (design.md §12.2). PICO_ATOM_UART_KEYS types what
    # arrives on UART1 at the guest (tools/uart-type.sh).
    option(PICO_ATOM_AUDIO "Audio output, and pacing on the PCM queue" ON)
    option(PICO_ATOM_UART_KEYS "Type characters received on UART1 at the guest" ON)

    # How much of the core's hot path runs from SRAM (src/core/hot.h).
    # Each tier was measured on the board; design.md §6.3 has the table
    # and says why this one ships.
    set(PICO_ATOM_RAM_TIER 2 CACHE STRING "Hot-code SRAM tier, 0-2 (src/core/hot.h)")

    # A tape put in the deck at boot, stopped, for a run driven over the
    # UART, where the menu cannot be reached (src/port/main.c). Empty in
    # a build that ships.
    set(PICO_ATOM_BOOT_TAPE "" CACHE STRING "Card path of a tape to insert at boot, or empty")
    if (PICO_ATOM_BOOT_TAPE)
        target_compile_definitions(pico-atom PRIVATE
            PICO_ATOM_BOOT_TAPE="${PICO_ATOM_BOOT_TAPE}")
    endif()

    # The same for a disc image in drive 0 (src/port/discio.h).
    set(PICO_ATOM_BOOT_DISC "" CACHE STRING "Card path of a disc for drive 0 at boot, or empty")
    if (PICO_ATOM_BOOT_DISC)
        target_compile_definitions(pico-atom PRIVATE
            PICO_ATOM_BOOT_DISC="${PICO_ATOM_BOOT_DISC}")
    endif()

    # clk_peri does not follow clk_sys by default; the SPI baud and the
    # audio carrier are re-applied in code as well (hardware-notes §3).
    target_compile_definitions(pico-atom PRIVATE
        PICO_ATOM_HAVE_FONT=${PICO_ATOM_HAVE_FONT}
        PICO_ATOM_AUDIO=$<BOOL:${PICO_ATOM_AUDIO}>
        PICO_ATOM_UART_KEYS=$<BOOL:${PICO_ATOM_UART_KEYS}>
        PICO_ATOM_RAM_TIER=${PICO_ATOM_RAM_TIER}
        PICO_CLOCK_ADJUST_PERI_CLOCK_WITH_SYS_CLOCK=1
        # UART1 at 115200 8-N-1, TX GP4 / RX GP5 (hardware-notes §2.7).
        PICO_DEFAULT_UART=1
        PICO_DEFAULT_UART_TX_PIN=4
        PICO_DEFAULT_UART_RX_PIN=5
    )

    target_link_libraries(pico-atom
        pico_atom_fatfs
        pico_stdlib
        pico_multicore
        pico_unique_id
        hardware_clocks
        hardware_dma
        hardware_i2c
        hardware_pwm
        hardware_spi
        hardware_sync
    )

    pico_enable_stdio_uart(pico-atom 1)
    pico_enable_stdio_usb(pico-atom 0)

    pico_add_extra_outputs(pico-atom)
endif()
